\section{Theoretical Foundation}
\textbf{Dittus-Boelter} (DITTUS_BOELTER): $Nu = 0.023 Re^{0.8} Pr^{0.4}$

\textit{Source:} Convection

\textbf{Fin Efficiency} (FIN_EFF): $\eta_f = \tanh(mL)/(mL), m=\sqrt{hP/(kA)}$

\textit{Source:} Extended surfaces


\section{Derivation Chain}
\subsection*{Step 1: Channel velocity}
\textbf{Inputs:} $\dot{V}, A_c$

\textbf{Formula:} $v = \dot{V} / (n A_c)$

\textbf{Principle:} Continuity --- m/s

\subsection*{Step 2: Reynolds number}
\textbf{Inputs:} $v, D_h, \nu$

\textbf{Formula:} $Re = v D_h / \nu$

\textbf{Principle:} Definition --- 

\subsection*{Step 3: Nusselt number}
\textbf{Inputs:} $Re, Pr$

\textbf{Formula:} $Nu = 0.023 Re^{0.8} Pr^{0.4}$

\textbf{Principle:} Dittus-Boelter --- Turbulent

\subsection*{Step 4: Heat transfer coeff}
\textbf{Inputs:} $Nu, k, D_h$

\textbf{Formula:} $h = Nu k / D_h$

\textbf{Principle:} Definition --- W/m²K

\subsection*{Step 5: Fin efficiency}
\textbf{Inputs:} $m, H$

\textbf{Formula:} $\eta_f = \tanh(mH)/(mH)$

\textbf{Principle:} Definition --- 

\subsection*{Step 6: Total area}
\textbf{Inputs:} $geometry$

\textbf{Formula:} $A_{tot} = A_b + n \eta_f A_f$

\textbf{Principle:} Effective --- m²

\subsection*{Step 7: Thermal resistance}
\textbf{Inputs:} $h, A$

\textbf{Formula:} $R_{th} = 1/(h A_{tot}) + t_b/(k A_b)$

\textbf{Principle:} Series --- K/W

\subsection*{Step 8: Pressure drop}
\textbf{Inputs:} $f, L, D_h, v$

\textbf{Formula:} $\Delta P = f (L/D_h) (\rho v^2/2)$

\textbf{Principle:} Darcy --- Pa
