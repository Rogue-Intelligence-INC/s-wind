\section{Theoretical Foundation}
\textbf{Drude Model} (DRUDE): $\varepsilon = 1 - \omega_p^2/(\omega^2 + i\gamma\omega)$

\textit{Source:} Metal optics

\textbf{Lorentz Oscillator} (LORENTZ): $\chi = \omega_p^2/(\omega_0^2 - \omega^2 - i\gamma\omega)$

\textit{Source:} Resonance

\subsection*{Step 4: Effective permeability}
\textbf{Inputs:} $f, f_0$

\textbf{Principle:} Lorentz --- Dispersive

\textbf{Inputs:} $f, f_p$

\textbf{Formula:} $\varepsilon_{eff} = 1 - f_p^2/(f^2 + i\gamma f)$

\textbf{Principle:} Drude --- Plasma
