\section{Derivation Chain}
\subsection*{Step 1: Osmotic pressure}
\textbf{Inputs:} $C, T$

\textbf{Formula:} $\pi = iCRT/M$

\textbf{Principle:} Van't Hoff --- bar

\subsection*{Step 2: Net driving pressure}
\textbf{Inputs:} $P, \pi_f, \pi_p$

\textbf{Formula:} $NDP = P - (\pi_m - \pi_p)$

\textbf{Principle:} Definition --- bar

\subsection*{Step 3: Permeate flux}
\textbf{Inputs:} $A_w, NDP$

\textbf{Formula:} $J_w = A_w \times NDP$

\textbf{Principle:} Solution-diffusion --- L/m²h

\subsection*{Step 4: Concentration polarization}
\textbf{Inputs:} $J_w, k$

\textbf{Formula:} $C_m/C_b = \exp(J_w/k)$

\textbf{Principle:} Film theory --- 

\subsection*{Step 5: Salt flux}
\textbf{Inputs:} $B, \Delta C$

\textbf{Formula:} $J_s = B(C_m - C_p)$

\textbf{Principle:} Diffusion --- g/m²h

\subsection*{Step 6: Salt rejection}
\textbf{Inputs:} $C_p, C_f$

\textbf{Formula:} $R = 1 - C_p/C_f$

\textbf{Principle:} Definition --- %

\subsection*{Step 7: Specific energy}
\textbf{Inputs:} $P, J_w, \eta$

\textbf{Formula:} $SEC = P/(J_w \times \eta)$

\textbf{Principle:} Definition --- kWh/m³

\subsection*{Step 8: Objective}
\textbf{Inputs:} $J_w, R$

\textbf{Formula:} $Z = J_w - \lambda\max(0, 0.995-R)$

\textbf{Principle:} Constrained --- Maximize
