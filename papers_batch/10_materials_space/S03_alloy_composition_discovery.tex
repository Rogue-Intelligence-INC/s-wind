\section{Theoretical Foundation}
\textbf{Hall-Petch} (HALL_PETCH): $\sigma_y = \sigma_0 + k_y/\sqrt{d}$

\textit{Source:} Hall, 1951

\textbf{Orowan Strengthening} (OROWAN): $\tau = Gb/L$

\textit{Source:} Orowan, 1948


\section{Derivation Chain}
\subsection*{Step 1: Solid solution}
\textbf{Inputs:} $x, \delta$

\textbf{Formula:} $\Delta\sigma_{ss} = k \sum_i x_i \delta_i^{4/3}$

\textbf{Principle:} Labusch --- MPa

\subsection*{Step 2: Precipitation}
\textbf{Inputs:} $f, r$

\textbf{Formula:} $\Delta\sigma_p = 0.81 M G b \sqrt{f/r} / (2\pi)$

\textbf{Principle:} Orowan --- MPa

\subsection*{Step 3: Grain boundary}
\textbf{Inputs:} $d$

\textbf{Formula:} $\Delta\sigma_{gb} = k_y / \sqrt{d}$

\textbf{Principle:} Hall-Petch --- MPa

\subsection*{Step 4: Total strength}
\textbf{Inputs:} $\Delta\sigma$

\textbf{Formula:} $\sigma_y = \sigma_0 + \sum \Delta\sigma_i^2)^{1/2}$

\textbf{Principle:} RSS --- MPa

\subsection*{Step 5: Density}
\textbf{Inputs:} $x, \rho$

\textbf{Formula:} $\rho = \sum_i x_i \rho_i / \sum_i x_i$

\textbf{Principle:} Rule of mixtures --- g/cm³

\subsection*{Step 6: Precipitate kinetics}
\textbf{Inputs:} $t, T$

\textbf{Formula:} $f = f_{eq}(1-e^{-k(T)t^n})$

\textbf{Principle:} JMAK --- Fraction

\subsection*{Step 7: Coarsening}
\textbf{Inputs:} $t, T$

\textbf{Formula:} $r = r_0 + k_r t^{1/3}$

\textbf{Principle:} LSW --- nm

\subsection*{Step 8: Specific strength}
\textbf{Inputs:} $\sigma_y, \rho$

\textbf{Formula:} $\sigma_y/\rho$

\textbf{Principle:} Objective --- MPa·cm³/g
