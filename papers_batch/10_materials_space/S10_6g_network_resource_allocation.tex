\section{Theoretical Foundation}
\textbf{Shannon Capacity} (SHANNON): $C = B\log_2(1+SNR)$

\textit{Source:} Shannon, 1948

\textbf{OFDM Spectral Efficiency} (OFDM): $SE = R \log_2(M) / (1+CP)$

\textit{Source:} Wireless communications


\section{Derivation Chain}
\subsection*{Step 1: Symbol duration}
\textbf{Inputs:} $\Delta f$

\textbf{Formula:} $T_s = 1/\Delta f$

\textbf{Principle:} Inverse --- μs

\subsection*{Step 2: Slot duration}
\textbf{Inputs:} $T_s, N_{symb}$

\textbf{Formula:} $T_{slot} = N_{symb} T_s$

\textbf{Principle:} Definition --- μs

\subsection*{Step 3: Throughput per layer}
\textbf{Inputs:} $M, R, \Delta f$

\textbf{Formula:} $R_l = M \times R \times \Delta f$

\textbf{Principle:} Modulation --- bits/s/Hz

\subsection*{Step 4: MIMO capacity}
\textbf{Inputs:} $L, SNR$

\textbf{Formula:} $C = L \log_2(1 + SNR/L)$

\textbf{Principle:} Shannon --- bits/s/Hz

\subsection*{Step 5: Spectral efficiency}
\textbf{Inputs:} $R_l, L$

\textbf{Formula:} $SE = \min(L \times R_l, C)$

\textbf{Principle:} Limited --- bits/s/Hz

\subsection*{Step 6: SNR}
\textbf{Inputs:} $P, N_0, BW$

\textbf{Formula:} $SNR = P / (N_0 \times BW)$

\textbf{Principle:} Definition --- 

\subsection*{Step 7: Latency}
\textbf{Inputs:} $T_{slot}, HARQ$

\textbf{Formula:} $t = T_{slot} \times (1 + r_{retx})$

\textbf{Principle:} Including retx --- ms

\subsection*{Step 8: Objective}
\textbf{Inputs:} $SE, t$

\textbf{Formula:} $Z = SE - \lambda\max(0, t-1)$

\textbf{Principle:} Constrained --- Maximize
