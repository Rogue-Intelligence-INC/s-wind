\section{Derivation Chain}
\subsection*{Step 1: Rule of mixtures}
\textbf{Inputs:} $E_f, E_m, V_f$

\textbf{Formula:} $E_1 = E_f V_f + E_m(1-V_f)$

\textbf{Principle:} Longitudinal --- GPa

\subsection*{Step 2: Transverse modulus}
\textbf{Inputs:} $E_f, E_m, V_f$

\textbf{Formula:} $E_2 = E_m/(1 - V_f(1-E_m/E_f))$

\textbf{Principle:} Halpin-Tsai --- GPa

\subsection*{Step 3: Transformed stiffness}
\textbf{Inputs:} $Q_{ij}, \theta$

\textbf{Formula:} $\bar{Q}_{ij} = T^{-1} Q T^{-T}$

\textbf{Principle:} Rotation --- GPa

\subsection*{Step 4: A matrix}
\textbf{Inputs:} $\bar{Q}, t$

\textbf{Formula:} $A_{ij} = \sum_k \bar{Q}_{ij}^{(k)} t_k$

\textbf{Principle:} Extensional --- N/m

\subsection*{Step 5: D matrix}
\textbf{Inputs:} $\bar{Q}, z, t$

\textbf{Formula:} $D_{ij} = \sum_k \bar{Q}_{ij}^{(k)} (z_k^3 - z_{k-1}^3)/3$

\textbf{Principle:} Bending --- Nm

\subsection*{Step 6: Effective modulus}
\textbf{Inputs:} $A, h$

\textbf{Formula:} $E_{eff} = A_{11}/(h)$

\textbf{Principle:} Normalized --- GPa

\subsection*{Step 7: Tsai-Wu criterion}
\textbf{Inputs:} $\sigma, F$

\textbf{Formula:} $F_1\sigma_1 + F_2\sigma_2 + F_{11}\sigma_1^2 + ... \leq 1$

\textbf{Principle:} Failure --- Constraint

\subsection*{Step 8: Specific stiffness}
\textbf{Inputs:} $E_{eff}, \rho$

\textbf{Formula:} $E/\rho = E_{eff}/(\rho_f V_f + \rho_m(1-V_f))$

\textbf{Principle:} Objective --- GPa·cm³/g
