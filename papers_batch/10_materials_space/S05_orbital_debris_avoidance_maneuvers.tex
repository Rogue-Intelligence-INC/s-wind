\section{Derivation Chain}
\subsection*{Step 1: Relative velocity}
\textbf{Inputs:} $v_1, v_2$

\textbf{Formula:} $v_{rel} = |\vec{v}_1 - \vec{v}_2|$

\textbf{Principle:} Definition --- km/s

\subsection*{Step 2: Time to closest approach}
\textbf{Inputs:} $r, v$

\textbf{Formula:} $TCA = t_0 + (\vec{r} \cdot \vec{v})/(v_{rel}^2)$

\textbf{Principle:} Linear --- s

\subsection*{Step 3: Miss distance}
\textbf{Inputs:} $r, v, TCA$

\textbf{Formula:} $d_{miss} = |\vec{r} + \vec{v}(TCA-t)|$

\textbf{Principle:} Propagation --- km

\subsection*{Step 4: Delta-v required}
\textbf{Inputs:} $d, t_w$

\textbf{Formula:} $\Delta v = d / t_w$

\textbf{Principle:} Impulsive --- m/s

\subsection*{Step 5: Fuel consumption}
\textbf{Inputs:} $\Delta v, I_{sp}$

\textbf{Formula:} $\Delta m = m(1-e^{-\Delta v/(g I_{sp})})$

\textbf{Principle:} Tsiolkovsky --- kg
