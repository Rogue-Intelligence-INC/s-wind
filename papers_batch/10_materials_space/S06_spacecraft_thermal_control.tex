\section{Derivation Chain}
\subsection*{Step 1: Solar input}
\textbf{Inputs:} $A, \alpha, S$

\textbf{Formula:} $Q_s = A \alpha S \cos\theta$

\textbf{Principle:} Absorption --- W

\subsection*{Step 2: Earth IR}
\textbf{Inputs:} $A, \varepsilon, F$

\textbf{Formula:} $Q_{IR} = A \varepsilon F_{E-S} \sigma T_E^4$

\textbf{Principle:} View factor --- W

\subsection*{Step 3: Albedo}
\textbf{Inputs:} $A, \alpha, a, F$

\textbf{Formula:} $Q_a = A \alpha a S F_{E-S}$

\textbf{Principle:} Reflection --- W

\subsection*{Step 4: Internal dissipation}
\textbf{Inputs:} $components$

\textbf{Formula:} $Q_i = \sum P_{elec}$

\textbf{Principle:} Electronics --- W

\subsection*{Step 5: Radiation out}
\textbf{Inputs:} $A_r, \varepsilon, T$

\textbf{Formula:} $Q_{out} = A_r \varepsilon \sigma T^4$

\textbf{Principle:} Stefan-Boltzmann --- W

\subsection*{Step 6: MLI conductance}
\textbf{Inputs:} $n$

\textbf{Formula:} $k_{MLI} = k_0 / n^{0.67}$

\textbf{Principle:} Empirical --- W/m²K

\subsection*{Step 7: Energy balance}
\textbf{Inputs:} $Q$

\textbf{Formula:} $Q_s + Q_{IR} + Q_a + Q_i + Q_h = Q_{out}$

\textbf{Principle:} Steady state --- W

\subsection*{Step 8: Heater power}
\textbf{Inputs:} $balance, T_{target}$

\textbf{Formula:} $Q_h = Q_{out} - Q_s - Q_{IR} - Q_a - Q_i$

\textbf{Principle:} Solve --- W
