\section{Theoretical Foundation}
\textbf{Swarm Intelligence} (SWARM): $\vec{v}_i = \vec{v}_i + c_1 r_1(\vec{p}_i-\vec{x}_i) + c_2 r_2(\vec{g}-\vec{x}_i)$ [Reynolds, 1987]

\textbf{Ground Sample Distance} (GSD): $GSD = (h \times sensor)/(f \times image)$ [Photogrammetry]


\section{Complete Derivation Chain}
\textbf{Step 1}: Swath width --- $h, FOV$ $\Rightarrow$ $W = 2h \tan(FOV/2)$ (Geometry: meters)


\textbf{Step 2}: Effective swath --- $W, o$ $\Rightarrow$ $W_{eff} = W(1-o)$ (Overlap removed: meters)


\textbf{Step 3}: Coverage rate single --- $v, W_{eff}$ $\Rightarrow$ $A_1 = v \times W_{eff}$ (Area/time: m²/s)


\textbf{Step 4}: Swarm coverage --- $N, A_1$ $\Rightarrow$ $A_N = N \times A_1 \times \eta_{coord}$ (Coordination: m²/s)


\textbf{Step 5}: GSD --- $h, f, p$ $\Rightarrow$ $GSD = h \times p / f$ (Resolution: cm/pixel)


\textbf{Step 6}: Flight time --- $E, P$ $\Rightarrow$ $t_f = E/(P_{hover} + P_{trans})$ (Battery: minutes)


\textbf{Step 7}: Coverage per charge --- $A_N, t_f$ $\Rightarrow$ $A_{max} = A_N \times t_f \times 60$ (Total: m²)


\textbf{Step 8}: Hourly throughput --- $A_{max}, t_c$ $\Rightarrow$ $Q = A_{max} \times 60/(t_f + t_c)$ (Including charge: m²/h)
