\section{Theoretical Foundation}
\textbf{Farquhar Model} (FARQUHAR): $A = \min(A_c, A_j) - R_d$ [Photosynthesis, 1980]

\textbf{Vapor Pressure Deficit} (VPD): $VPD = e_s(T)(1-RH/100)$ [Plant physiology]


\section{Complete Derivation Chain}
\textbf{Step 1}: Heat balance --- $Q_{in}, Q_{out}$ $\Rightarrow$ $dT/dt = (Q_{solar} + Q_{heat} - Q_{vent} - Q_{loss})/(\rho C_p V)$ (Energy: °C/s)


\textbf{Step 2}: Solar gain --- $G, A, \tau$ $\Rightarrow$ $Q_{solar} = G \times A \times \tau$ (Transmission: W)


\textbf{Step 3}: Heat loss --- $U, A, \Delta T$ $\Rightarrow$ $Q_{loss} = U A (T_{in} - T_{out})$ (Conduction: W)


\textbf{Step 4}: Ventilation load --- $V, \rho, c_p, \Delta T$ $\Rightarrow$ $Q_{vent} = V \rho c_p \Delta T$ (Air exchange: W)


\textbf{Step 5}: Humidity balance --- $E_t, V_{in}, V_{out}$ $\Rightarrow$ $d\omega/dt = (E_t - V(\omega - \omega_{out}))/V$ (Moisture: kg/s)


\textbf{Step 6}: CO2 uptake --- $P_n$ $\Rightarrow$ $dC/dt = (C_{inject} - P_n - V(C-C_{out}))/V$ (Carbon: ppm/s)


\textbf{Step 7}: Photosynthesis rate --- $PAR, C, T$ $\Rightarrow$ $P_n = P_{max} \times f(PAR) \times f(C) \times f(T)$ (Farquhar: μmol/m²s)


\textbf{Step 8}: Energy cost --- $Q_{heat}, C_{inject}, prices$ $\Rightarrow$ $Cost = Q_{heat} p_e + C_{inject} p_C$ (Operating: $/h)
