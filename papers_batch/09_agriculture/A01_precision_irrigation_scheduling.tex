\section{Theoretical Foundation}
\textbf{Penman-Monteith} (FAO56): $ET_0 = \frac{0.408\Delta(R_n-G)+\gamma\frac{900}{T+273}u_2(e_s-e_a)}{\Delta+\gamma(1+0.34u_2)}$ [FAO, 1998]

\textbf{Yield-Water Relation} (STEWART): $1-Y/Y_m = K_y(1-ET_a/ET_m)$ [Stewart, 1977]


\section{Complete Derivation Chain}
\textbf{Step 1}: Soil water balance --- $P, I, ET, D$ $\Rightarrow$ $\theta_{t+1} = \theta_t + (P + I - ET - D)/Z_r$ (Conservation: Mass balance)


\textbf{Step 2}: ET estimation --- $ET_0, K_c$ $\Rightarrow$ $ET_c = K_c \times ET_0$ (FAO-56: Crop ET)


\textbf{Step 3}: Deep percolation --- $\theta, \theta_{FC}$ $\Rightarrow$ $D = K_s (\theta/\theta_{sat})^n$ (Darcy: Drainage)


\textbf{Step 4}: Water stress factor --- $\theta, \theta_{wp}, \theta_{FC}$ $\Rightarrow$ $K_s = (\theta - \theta_{wp})/(\theta_{FC} - \theta_{wp})$ (Linear: Reduction)


\textbf{Step 5}: Yield response --- $ET_a, ET_c$ $\Rightarrow$ $Y/Y_m = 1 - K_y(1 - ET_a/ET_c)$ (FAO: Stewart)


\textbf{Step 6}: Water use efficiency --- $Y, I$ $\Rightarrow$ $WUE = Y/I$ (Definition: kg/m³)


\textbf{Step 7}: Irrigation requirement --- $ET_c, eff$ $\Rightarrow$ $I_r = ET_c / \eta_{irr}$ (Efficiency: Gross)


\textbf{Step 8}: Objective --- $Y, I$ $\Rightarrow$ $Z = Y/I$ (WUE: Maximize)
