\section{Theoretical Foundation}
\textbf{Thermal Death} (ARRHENIUS): $D = D_0 e^{-E_a/(RT)}$ [First-order kinetics]

\textbf{F-value} (FVALUE): $F_0 = \int 10^{(T-121.1)/10} dt$ [Canning, Ball 1923]


\section{Complete Derivation Chain}
\textbf{Step 1}: Log reduction --- $D, t$ $\Rightarrow$ $LR = t/D_T$ (Definition: Log CFU)


\textbf{Step 2}: D-value temperature --- $D, T, z$ $\Rightarrow$ $D_T = D_{ref} \times 10^{(T_{ref}-T)/z}$ (First order: seconds)


\textbf{Step 3}: F-value --- $T, z$ $\Rightarrow$ $F = \int 10^{(T(t)-T_{ref})/z} dt$ (Lethality: equivalent)


\textbf{Step 4}: Nutrient retention --- $D_n, t$ $\Rightarrow$ $R = 10^{-t/D_n}$ (First order: Fraction)


\textbf{Step 5}: Cook value --- $T, z_c$ $\Rightarrow$ $C = \int 10^{(T(t)-100)/z_c} dt$ (Quality: equivalent)


\textbf{Step 6}: Energy input --- $Q, c_p, \Delta T$ $\Rightarrow$ $E = Q \rho c_p \Delta T$ (Heating: kW)


\textbf{Step 7}: Regeneration --- $r, E$ $\Rightarrow$ $E_{net} = E(1-r)$ (Recovery: kW)


\textbf{Step 8}: Process objective --- $LR, R$ $\Rightarrow$ $Z = LR - \lambda \max(0, 0.8-R)$ (Constrained: Maximize)
