\section{Theoretical Foundation}
\textbf{Fick's Diffusion} (FICK): $J = -D \nabla C$ [Mass transfer]

\textbf{Gamma Model} (GAMMA): $\mu = \mu_{max} \prod_i \gamma_i$ [Predictive microbiology]


\section{Complete Derivation Chain}
\textbf{Step 1}: O2 transmission --- $P, A, t$ $\Rightarrow$ $OTR = P_{O2} \times A / t$ (Fick's law: cc/day)


\textbf{Step 2}: Respiration rate --- $T, O2$ $\Rightarrow$ $R = R_0 \times Q_{10}^{(T-T_0)/10} \times O_2/(K_m+O_2)$ (Michaelis: cc/kg/h)


\textbf{Step 3}: O2 equilibrium --- $OTR, R, m$ $\Rightarrow$ $O_{2,eq} = O_{2,atm} - R \times m / OTR$ (Steady state: %)


\textbf{Step 4}: Microbial growth --- $T, a_w, O2$ $\Rightarrow$ $\mu = \mu_{max} \gamma(T) \gamma(a_w) \gamma(O_2)$ (Gamma model: 1/h)


\textbf{Step 5}: Shelf life --- $N_0, N_{max}, \mu$ $\Rightarrow$ $SL = \ln(N_{max}/N_0)/\mu$ (Log growth: hours)


\textbf{Step 6}: Quality decay --- $k, T$ $\Rightarrow$ $Q = Q_0 e^{-kt}$ (First order: Score)


\textbf{Step 7}: Film cost --- $t, material$ $\Rightarrow$ $C_f = t \times \rho \times price$ (Linear: $/m²)


\textbf{Step 8}: Package cost --- $C_f, A, AS$ $\Rightarrow$ $C_p = C_f \times A + C_{AS}$ (Total: $/unit)
