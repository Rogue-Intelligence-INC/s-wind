\section{Theoretical Foundation}
\textbf{Growing Degree Days} (GDD): $GDD = \sum(T_{mean} - T_{base})$ [Agriculture]

\textbf{Vegetation Index} (NDVI): $NDVI = (NIR-RED)/(NIR+RED)$ [Remote Sensing]


\section{Complete Derivation Chain}
\textbf{Step 1}: NDVI integration --- $NDVI(t)$ $\Rightarrow$ $\bar{NDVI} = \int_{t_1}^{t_2} NDVI(t) dt / (t_2-t_1)$ (Average: Season)


\textbf{Step 2}: GDD accumulation --- $T$ $\Rightarrow$ $GDD = \sum \max(0, (T_{max}+T_{min})/2 - T_b)$ (Thermal units: °C-day)


\textbf{Step 3}: Precipitation index --- $P$ $\Rightarrow$ $P_{eff} = \sum P_i \times e^{-i/\tau}$ (Weighted: Decay)


\textbf{Step 4}: Combined index --- $w, vars$ $\Rightarrow$ $CI = w_N \bar{NDVI} + w_T GDD + w_P P_{eff}$ (Linear: Composite)


\textbf{Step 5}: Yield model --- $CI$ $\Rightarrow$ $Y = a + b \times CI$ (Regression: Prediction)


\textbf{Step 6}: Residual --- $Y_{obs}, Y_{pred}$ $\Rightarrow$ $e = Y_{obs} - Y_{pred}$ (Error: Per sample)


\textbf{Step 7}: RMSE --- $e$ $\Rightarrow$ $RMSE = \sqrt{\sum e_i^2/n}$ (Metric: Accuracy)


\textbf{Step 8}: R-squared --- $SS_{res}, SS_{tot}$ $\Rightarrow$ $R^2 = 1 - SS_{res}/SS_{tot}$ (Goodness: Fit)
