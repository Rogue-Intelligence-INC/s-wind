\section{Theoretical Foundation}
\textbf{McCree Curve} (McCREE): $PPFD = \int_{400}^{700} \Phi(\lambda) d\lambda$ [Plant action spectrum]

\textbf{Emerson Enhancement} (EMERSON): $P_{R+FR} > P_R + P_{FR}$ [Red/far-red synergy]


\section{Complete Derivation Chain}
\textbf{Step 1}: DLI --- $I, P$ $\Rightarrow$ $DLI = I \times P \times 3600 / 10^6$ (Daily light integral: mol/m²d)


\textbf{Step 2}: Blue absorption --- $f_B, I$ $\Rightarrow$ $A_B = f_B I \times \phi_B$ (Quantum yield: μmol/m²s)


\textbf{Step 3}: Red absorption --- $f_R, I$ $\Rightarrow$ $A_R = f_R I \times \phi_R$ (Quantum yield: μmol/m²s)


\textbf{Step 4}: PSS ratio --- $f_R, f_{FR}$ $\Rightarrow$ $PSS = 0.89 f_R / (f_R + f_{FR})$ (Phytochrome: Ratio)


\textbf{Step 5}: Photosynthesis rate --- $A_B, A_R$ $\Rightarrow$ $P_n = P_{max} (1-e^{-k(A_B+A_R)/P_{max}})$ (Saturation: μmol CO2/m²s)


\textbf{Step 6}: Electrical power --- $I, eff$ $\Rightarrow$ $P_e = I \times A / (\eta_{LED} \times 4.6)$ (Conversion: W)


\textbf{Step 7}: Photon efficacy --- $I, P_e$ $\Rightarrow$ $\eta_{ph} = I / P_e$ (Definition: μmol/J)


\textbf{Step 8}: Objective --- $P_n, P_e$ $\Rightarrow$ $Z = P_n / P_e$ (Efficiency: Maximize)
