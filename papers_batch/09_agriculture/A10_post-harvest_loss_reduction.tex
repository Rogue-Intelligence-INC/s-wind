\section{Theoretical Foundation}
\textbf{Q10 Temperature Coefficient} (Q10): $Q_{10} = (R_2/R_1)^{10/(T_2-T_1)}$ [Biochemistry]

\textbf{Vapor Pressure Deficit} (VPD): $VPD = e_s(T)(1-RH/100)$ [Storage science]


\section{Complete Derivation Chain}
\textbf{Step 1}: Respiration loss --- $T$ $\Rightarrow$ $L_r = R_0 \times Q_{10}^{(T-T_0)/10} \times t$ (Mass loss: %)


\textbf{Step 2}: Transpiration loss --- $VPD, k$ $\Rightarrow$ $L_t = k \times VPD \times t$ (Water loss: %)


\textbf{Step 3}: Mechanical damage --- $h$ $\Rightarrow$ $L_m = a \times h^2$ (Compression: %)


\textbf{Step 4}: Decay loss --- $T, RH, t$ $\Rightarrow$ $L_d = \mu_{pathogen} \times t$ (Microbial: %)


\textbf{Step 5}: Sorting removal --- $S, quality$ $\Rightarrow$ $R_s = S \times P(defect)$ (Initial cull: %)


\textbf{Step 6}: Net loss --- $L$ $\Rightarrow$ $L_{total} = L_r + L_t + L_m + L_d - R_s$ (Sum: %)


\textbf{Step 7}: Energy cost --- $T, T_{amb}$ $\Rightarrow$ $C_e = (T_{amb}-T) \times k_{cool} \times t$ (Cooling: $/kg)


\textbf{Step 8}: Total cost --- $C_e, C_s, L$ $\Rightarrow$ $C = C_e + C_s + L \times price$ (Economic: $/kg)
