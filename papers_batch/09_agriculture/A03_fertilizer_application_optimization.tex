\section{Theoretical Foundation}
\textbf{Liebig's Law} (LIEBIG): $Y = \min_i(Y_i(N_i))$ [Minimum factor]

\textbf{Mitscherlich} (MITSCHERLICH): $Y = A(1-e^{-cx})$ [Diminishing returns, 1909]


\section{Complete Derivation Chain}
\textbf{Step 1}: Mineralization --- $SOM, T$ $\Rightarrow$ $N_m = k \times SOM \times Q_{10}^{(T-25)/10}$ (Arrhenius: kg/ha)


\textbf{Step 2}: Plant uptake --- $N_{avail}, LAI$ $\Rightarrow$ $U = N_{avail} \times (1-e^{-k_{ext} LAI})$ (Michaelis: kg/ha)


\textbf{Step 3}: Leaching --- $N_{avail}, P$ $\Rightarrow$ $L = N_{avail} \times (P/P_{crit})^2$ (Exponential: kg/ha)


\textbf{Step 4}: Volatilization --- $N, pH, T$ $\Rightarrow$ $V = N \times f(pH) \times Q_{10}^{(T-20)/10}$ (Empirical: kg/ha)


\textbf{Step 5}: N balance --- $inputs, outputs$ $\Rightarrow$ $\Delta N = (N + N_m) - (U + L + V)$ (Conservation: kg/ha)


\textbf{Step 6}: Yield response --- $U$ $\Rightarrow$ $Y = Y_0 + a U - b U^2$ (Quadratic: kg/ha)


\textbf{Step 7}: NUE --- $Y, N$ $\Rightarrow$ $NUE = (Y - Y_0)/N$ (Agronomic: kg/kg)


\textbf{Step 8}: Environmental penalty --- $L, V$ $\Rightarrow$ $P_e = c_L L + c_V V$ (Cost: $/ha)
