\textbf{Principle:} Definition --- Material removal

\subsection*{Step 2: Theoretical roughness}
\textbf{Inputs:} $G$

\textbf{Formula:} $R_{a,th} = k / G^{1.5}$

\subsection*{Step 3: Speed correction}
\textbf{Inputs:} $v_s$

\textbf{Formula:} $f_v = (v_s/30)^{-0.3}$

\textbf{Principle:} Power law --- Higher speed = better

\subsection*{Step 4: Infeed correction}
\textbf{Inputs:} $a_e$

\textbf{Formula:} $f_a = (a_e/0.02)^{0.5}$

\textbf{Principle:} Power law --- Deeper = rougher

\subsection*{Step 5: Coolant correction}
\textbf{Inputs:} $P_c$

\textbf{Formula:} $f_c = (P_c/10)^{-0.1}$

\textbf{Principle:} Weak effect --- Thermal

\subsection*{Step 6: Actual roughness}
\textbf{Inputs:} $R_{a,th}, f_v, f_a, f_c$

\textbf{Formula:} $R_a = R_{a,th} f_v f_a f_c$

\textbf{Principle:} Product --- Objective

\subsection*{Step 7: Material removal rate}
\textbf{Inputs:} $a_e, v_t, w$

\textbf{Formula:} $MRR = a_e \times v_t \times w$

\textbf{Principle:} Definition --- mm³/min

\subsection*{Step 8: Process time}
\textbf{Inputs:} $V, MRR$

\textbf{Formula:} $T = V/MRR$

\textbf{Principle:} Definition --- Constraint via cost

\subsection*{Step 9: Cost}
\textbf{Inputs:} $T, r$

\textbf{Formula:} $C = T \times r$

\textbf{Principle:} Time-based --- Budget limit
