\section{Derivation Chain}

The complete derivation proceeds through 8 sequential steps:

\subsection*{Step 1: Path length}
\textbf{Inputs:} $waypoints$

\textbf{Formula:} $L = \sum_i \|p_{i+1} - p_i\|$

\textbf{Principle:} Euclidean --- Total distance

\subsection*{Step 2: Time-optimal velocity}
\textbf{Inputs:} $a, v$

\textbf{Formula:} $v^* = \min(v_{max}, \sqrt{2a_{max}\Delta s})$

\textbf{Principle:} Kinematics --- Bounded

\subsection*{Step 3: Acceleration phase}
\textbf{Inputs:} $v^*, a_{max}$

\textbf{Formula:} $t_a = v^*/a_{max}$

\textbf{Principle:} Definition --- Ramp up

\subsection*{Step 4: Cruise phase}
\textbf{Inputs:} $L, v^*, t_a$

\textbf{Formula:} $t_c = (L - v^* t_a)/v^*$

\textbf{Principle:} Remaining --- Constant speed

\subsection*{Step 5: Deceleration phase}
\textbf{Inputs:} $v^*, a_{max}$

\textbf{Formula:} $t_d = v^*/a_{max}$

\textbf{Principle:} Symmetric --- Ramp down

\subsection*{Step 6: Cycle time}
\textbf{Inputs:} $t_a, t_c, t_d$

\textbf{Formula:} $T = t_a + t_c + t_d$

\textbf{Principle:} Total --- Objective

\subsection*{Step 7: Joint torque}
\textbf{Inputs:} $M, \ddot{q}$

\textbf{Formula:} $\tau = M(q)\ddot{q} + C(q,\dot{q})\dot{q} + g(q)$

\textbf{Principle:} Dynamics --- Constraint

\subsection*{Step 8: Collision distance}
\textbf{Inputs:} $p, obstacles$

\textbf{Formula:} $d_{min} = \min_i \|p - o_i\|$

\textbf{Principle:} Minimum --- Must exceed d_s


\section{Exhaustive Search Algorithm}

\begin{verbatim}

    best_value = -infinity
