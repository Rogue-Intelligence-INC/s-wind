\subsection{Knowledge Base References}

\section{Derivation Chain}

The complete derivation proceeds through 8 sequential steps:

\subsection*{Step 1: Probability of acceptance}
\textbf{Inputs:} $n, c, p$

\textbf{Formula:} $P_a = \sum_{i=0}^{c} \binom{n}{i} p^i (1-p)^{n-i}$

\textbf{Principle:} Binomial --- OC curve

\subsection*{Step 2: Average outgoing quality}
\textbf{Inputs:} $p, P_a, n, N$

\textbf{Formula:} $AOQ = p P_a (N-n)/N$

\textbf{Principle:} Definition --- After inspection

\subsection*{Step 3: Average total inspection}
\textbf{Inputs:} $n, N, P_a$

\textbf{Formula:} $ATI = n + (1-P_a)(N-n)$

\textbf{Principle:} Expected --- 100% on reject

\subsection*{Step 4: Inspection cost}
\textbf{Inputs:} $ATI, c_i$

\textbf{Formula:} $C_i = ATI \times c_i$

\textbf{Principle:} Unit cost --- Direct

\subsection*{Step 5: Defect escape cost}
\textbf{Inputs:} $AOQ, N, c_d$

\textbf{Formula:} $C_d = AOQ \times N \times c_d$

\textbf{Principle:} Expected --- Field failures

\subsection*{Step 6: Total cost}
\textbf{Inputs:} $C_i, C_d$

\textbf{Formula:} $C_{tot} = C_i + C_d$

\textbf{Principle:} Sum --- Minimize

\subsection*{Step 7: Consumer's risk}
\textbf{Inputs:} $p_2, P_a$

\textbf{Formula:} $\beta = P_a(p_2)$

\textbf{Principle:} LTPD --- Constraint

\subsection*{Step 8: Producer's risk}
\textbf{Inputs:} $p_1, P_a$

\textbf{Formula:} $\alpha = 1 - P_a(p_1)$

\textbf{Principle:} AQL --- Type I error


\section{Exhaustive Search Algorithm}

    best_value = -infinity
