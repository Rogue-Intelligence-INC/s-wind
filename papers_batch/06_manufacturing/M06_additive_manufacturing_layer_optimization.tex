\subsection*{Step 1: Number of layers}

\textbf{Principle:} Ceiling --- Total layers

\subsection*{Step 2: Perimeter length}
\textbf{Inputs:} $geometry$

\textbf{Formula:} $L_p = 2\pi r + 4(l+w)$

\textbf{Principle:} Approximation --- Outer shell

\subsection*{Step 3: Infill path length}
\textbf{Inputs:} $A, \rho, w_e$

\textbf{Formula:} $L_i = A \rho / w_e$

\textbf{Principle:} Area fill --- Interior

\subsection*{Step 4: Layer time}
\textbf{Inputs:} $L_p, L_i, v$

\textbf{Formula:} $t_L = (L_p + L_i)/v$

\textbf{Principle:} Definition --- Per layer

\subsection*{Step 5: Print time}
\textbf{Inputs:} $N_L, t_L, t_{move}$

\textbf{Formula:} $T = N_L(t_L + t_{move})$

\textbf{Principle:} Total --- Objective

\subsection*{Step 6: Staircase error}
\textbf{Inputs:} $h, \theta$

\textbf{Formula:} $e_s = h \tan\theta / 2$

\textbf{Principle:} Geometry --- Surface deviation

\subsection*{Step 7: Thermal shrinkage}
\textbf{Inputs:} $\alpha, T_n, T_b$

\textbf{Formula:} $\delta = \alpha L (T_n - T_b)$

\textbf{Principle:} Linear --- Dimensional

\subsection*{Step 8: Total error}
\textbf{Inputs:} $e_s, \delta$

\textbf{Formula:} $e_{tot} = \sqrt{e_s^2 + \delta^2}$

\textbf{Principle:} RSS --- Constraint
