\begin{abstract}
This paper presents exhaustive parametric optimization for smart grid load balancing using the S-wind simulation framework.
We systematically explore a search space of 20,736 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Minimize peak load subject to user comfort constraints}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\section{Derivation Chain}

\subsection*{Step 1: Base load profile}
\textbf{Inputs:} $t$

$L_0(t) = \sum_i l_i(t)$

\textbf{Principle:} Aggregation --- Uncontrolled

\subsection*{Step 2: Flexible load}
\textbf{Inputs:} $L_0, \alpha$

$L_f(t) = \alpha L_0(t) \mathbf{1}_{shiftable}$

\textbf{Principle:} Subset --- DR eligible

\subsection*{Step 3: Shifted load}
\textbf{Inputs:} $L_f, \Delta t$

$L_s(t) = \sum_{\tau} w(\tau) L_f(t-\tau)$

\textbf{Principle:} Convolution --- Redistributed

\subsection*{Step 4: New profile}
\textbf{Inputs:} $L_0, L_f, L_s$

$L(t) = L_0(t) - L_f(t) + L_s(t)$

\textbf{Principle:} Balance --- Modified

\subsection*{Step 5: Peak load}
\textbf{Inputs:} $L(t)$

$L_{peak} = \max_t L(t)$

\textbf{Principle:} Definition --- Minimize

\subsection*{Step 6: Load factor}
\textbf{Inputs:} $L_{avg}, L_{peak}$

$LF = L_{avg}/L_{peak}$

\textbf{Principle:} Definition --- Improve

\subsection*{Step 7: Comfort penalty}
\textbf{Inputs:} $delays$

$P_c = \sum_i c_i |\Delta t_i|^2$

\textbf{Principle:} Quadratic --- Constraint

\subsection*{Step 8: Cost savings}
\textbf{Inputs:} $L_{peak}, \Delta L$

$S = \Delta L_{peak} \times C_{cap}$

\textbf{Principle:} Capacity --- Benefit
