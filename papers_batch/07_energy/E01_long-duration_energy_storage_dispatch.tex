\begin{abstract}
This paper presents exhaustive parametric optimization for long-duration energy storage dispatch using the S-wind simulation framework.
We systematically explore a search space of 726,000 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Maximize revenue from arbitrage subject to state-of-charge limits}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\section{Derivation Chain}

\subsection*{Step 1: Energy throughput}
\textbf{Inputs:} $P_{max}, t_c$

$E_c = P_{max} \times t_c$

\textbf{Principle:} Definition --- Charge energy

\subsection*{Step 2: Discharge energy}
\textbf{Inputs:} $E_c, \eta$

$E_d = E_c \times \eta$

\textbf{Principle:} Efficiency loss --- Deliverable

\subsection*{Step 3: Price spread}
\textbf{Inputs:} $p_{peak}, p_{off}$

$\Delta p = p_{peak} - p_{off}/\eta$

\textbf{Principle:} Net margin --- Per MWh

\subsection*{Step 4: Daily revenue}
\textbf{Inputs:} $E_d, \Delta p$

$R = E_d \times \Delta p$

\textbf{Principle:} Definition --- Gross

\subsection*{Step 5: Degradation cost}
\textbf{Inputs:} $cycles, C_{cap}$

$C_{deg} = C_{cap}/N_{life}$

\textbf{Principle:} Per cycle --- Amortized

\subsection*{Step 6: Net revenue}
\textbf{Inputs:} $R, C_{deg}$

$R_{net} = R - C_{deg}$

\textbf{Principle:} Objective --- Maximize

\subsection*{Step 7: SOC constraint}
\textbf{Inputs:} $E, E_{max}$

$0.1 E_{max} \leq E \leq 0.9 E_{max}$

\textbf{Principle:} Bounds --- Battery life

\subsection*{Step 8: Annual revenue}
\textbf{Inputs:} $R_{net}, days$

$R_{annual} = R_{net} \times 365$

\textbf{Principle:} Scaling --- Year
