\begin{abstract}
This paper presents exhaustive parametric optimization for pumped hydro storage scheduling using the S-wind simulation framework.
We systematically explore a search space of 104,000 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Maximize arbitrage profit subject to reservoir and turbine constraints}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\section{Derivation Chain}

\subsection*{Step 1: Potential energy}
\textbf{Inputs:} $V, \rho, g, H$

$E = \rho g H V$

\textbf{Principle:} Definition --- Stored energy

\subsection*{Step 2: Storage capacity}
\textbf{Inputs:} $E$

$E_{max} = \rho g H V_u / 3.6e9$

\textbf{Principle:} Conversion --- MWh

\subsection*{Step 3: Pump power}
\textbf{Inputs:} $Q_p, H, \eta_p$

$P_p = \rho g H Q_p / \eta_p$

\textbf{Principle:} Hydraulics --- Input

\subsection*{Step 4: Turbine power}
\textbf{Inputs:} $Q_t, H, \eta_t$

$P_t = \rho g H Q_t \eta_t$

\textbf{Principle:} Hydraulics --- Output

\subsection*{Step 5: Volume balance}
\textbf{Inputs:} $Q_p, Q_t, V$

$dV/dt = Q_p - Q_t$

\textbf{Principle:} Conservation --- Reservoir level

\subsection*{Step 6: Revenue}
\textbf{Inputs:} $P_t, p_{sell}$

$R = \sum_t P_t(t) p(t) \Delta t$

\textbf{Principle:} Integration --- Discharge

\subsection*{Step 7: Cost}
\textbf{Inputs:} $P_p, p_{buy}$

$C = \sum_t P_p(t) p(t) / \eta \Delta t$

\textbf{Principle:} Integration --- Charge

\subsection*{Step 8: Profit}
\textbf{Inputs:} $R, C$

$\Pi = R - C$

\textbf{Principle:} Definition --- Maximize
