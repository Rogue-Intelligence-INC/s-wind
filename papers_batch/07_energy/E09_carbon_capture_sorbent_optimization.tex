\begin{abstract}
This paper presents exhaustive parametric optimization for carbon capture sorbent optimization using the S-wind simulation framework.
We systematically explore a search space of 182,400 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Maximize CO2 capture capacity per regeneration energy}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\section{Theoretical Foundation}

\subsection{Knowledge Base References}
\textbf{Langmuir Isotherm} (LANGMUIR): $q = q_m K p / (1 + K p)$ --- Langmuir, 1918

\textbf{Van't Hoff Equation} (VANTHOFF): $d\ln K/dT = \Delta H/(RT^2)$ --- Van't Hoff, 1884


\section{Derivation Chain}

\subsection*{Step 1: CO2 adsorption}
\textbf{Inputs:} $p, T, K$

$q = q_m K p / (1 + K p)$

\textbf{Principle:} Langmuir --- Equilibrium

\subsection*{Step 2: Saturation capacity}
\textbf{Inputs:} $S, L_a$

$q_m = k_s S L_a$

\textbf{Principle:} Proportional --- Maximum

\subsection*{Step 3: Heat of adsorption}
\textbf{Inputs:} $T, K$

$\Delta H = R T^2 (\partial \ln K / \partial T)$

\textbf{Principle:} Van't Hoff --- Enthalpy

\subsection*{Step 4: Regeneration energy}
\textbf{Inputs:} $q, \Delta H, C_p, \Delta T$

$E_r = q|\Delta H| + m C_p \Delta T$

\textbf{Principle:} Sum --- Total heat

\subsection*{Step 5: Working capacity}
\textbf{Inputs:} $q_{ads}, q_{des}$

$\Delta q = q(T_{ads}) - q(T_{des})$

\textbf{Principle:} Swing --- Useful

\subsection*{Step 6: Capture rate}
\textbf{Inputs:} $\Delta q, t_c$

$r = \Delta q / t_c$

\textbf{Principle:} Definition --- kg/h

\subsection*{Step 7: Energy efficiency}
\textbf{Inputs:} $\Delta q, E_r$

$\eta_E = \Delta q / E_r$

\textbf{Principle:} Definition --- Maximize

\subsection*{Step 8: Productivity}
\textbf{Inputs:} $r, mass$

$P = r \times mass / volume$

\textbf{Principle:} Specific --- Process
