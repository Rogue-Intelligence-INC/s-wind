\begin{abstract}
This paper presents exhaustive parametric optimization for grid-scale battery degradation management using the S-wind simulation framework.
We systematically explore a search space of 61,952 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Maximize battery lifetime value subject to grid service requirements}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\section{Derivation Chain}

\subsection*{Step 1: Cycle stress factor}
\textbf{Inputs:} $DOD$

$\sigma_c = (DOD/0.8)^{1.5}$

\textbf{Principle:} Empirical --- Depth impact

\subsection*{Step 2: Rate stress factor}
\textbf{Inputs:} $C$

$\sigma_r = (C/1)^{0.8}$

\textbf{Principle:} Power law --- Current impact

\subsection*{Step 3: Temperature factor}
\textbf{Inputs:} $T$

$\sigma_T = e^{(T-25)/10}$

\textbf{Principle:} Arrhenius --- Thermal aging

\subsection*{Step 4: SOC stress}
\textbf{Inputs:} $SOC_m$

$\sigma_s = 1 + 0.5|SOC_m - 0.5|$

\textbf{Principle:} Calendar aging --- High/low SOC

\subsection*{Step 5: Combined stress}
\textbf{Inputs:} $\sigma$

$S = \sigma_c \sigma_r \sigma_T \sigma_s$

\textbf{Principle:} Product --- Total

\subsection*{Step 6: Cycle life}
\textbf{Inputs:} $N_0, S$

$N = N_0 / S$

\textbf{Principle:} Degradation --- Expected

\subsection*{Step 7: Energy throughput}
\textbf{Inputs:} $N, E, DOD$

$E_{life} = N \times E \times DOD$

\textbf{Principle:} Definition --- Total

\subsection*{Step 8: Lifetime value}
\textbf{Inputs:} $E_{life}, margin$

$V = E_{life} \times margin - CAPEX$

\textbf{Principle:} NPV --- Objective
