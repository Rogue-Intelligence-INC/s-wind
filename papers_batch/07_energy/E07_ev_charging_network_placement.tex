\begin{abstract}
This paper presents exhaustive parametric optimization for ev charging network placement using the S-wind simulation framework.
We systematically explore a search space of 37,800 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Minimize total user waiting time subject to investment budget}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\subsection{Knowledge Base References}
\textbf{Erlang-C Formula} (ERLANG_C): $P(wait) = \frac{(n\rho)^n}{n!(1-\rho)}P_0$ --- Erlang, 1917

\textbf{Little's Law} (LITTLE): $L = \lambda W$ --- Little, 1961


\section{Derivation Chain}

\subsection*{Step 1: Demand density}
\textbf{Inputs:} $location$

$\lambda(x) = \rho_{EV} v_{avg} / d_{range}$

\textbf{Principle:} Traffic model --- Arrivals/km²

\subsection*{Step 2: Station demand}
\textbf{Inputs:} $\lambda, A_s$

$\Lambda_s = \int_{A_s} \lambda(x) dx$

\textbf{Principle:} Integration --- Arrivals/h

\subsection*{Step 3: Service rate}
\textbf{Inputs:} $P, E_{avg}$

$\mu = P / E_{avg}$

\textbf{Principle:} Definition --- Vehicles/h

\subsection*{Step 4: Utilization}
\textbf{Inputs:} $\Lambda, \mu, n_c$

$\rho = \Lambda / (n_c \mu)$

\textbf{Principle:} M/M/c --- Load factor

\subsection*{Step 5: Queue length}
\textbf{Inputs:} $\rho, n_c$

$L_q = P_0 \rho^{n_c} \rho / (n_c!(1-\rho)^2)$

\textbf{Principle:} Erlang-C --- Waiting

\subsection*{Step 6: Wait time}
\textbf{Inputs:} $L_q, \Lambda$

$W_q = L_q / \Lambda$

\textbf{Principle:} Little's law --- Per user

\subsection*{Step 7: Total wait}
\textbf{Inputs:} $W_q, \Lambda$

$W_{tot} = \sum_s W_{q,s} \Lambda_s$

\textbf{Principle:} Sum --- Objective

\subsection*{Step 8: Investment}
\textbf{Inputs:} $N_s, n_c, P$

$I = N_s(I_{site} + n_c I_c(P))$

\textbf{Principle:} Sum --- Budget constraint
