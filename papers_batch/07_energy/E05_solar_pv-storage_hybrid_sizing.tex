\begin{abstract}
This paper presents exhaustive parametric optimization for solar pv-storage hybrid sizing using the S-wind simulation framework.
We systematically explore a search space of 3,380,000 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Minimize LCOE for off-grid system with 99.9% reliability}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\section{Theoretical Foundation}

\subsection{Knowledge Base References}
\textbf{Perez Irradiance Model} (PEREZ): $G_t = G_b R_b + G_d R_d + G\rho R_r$ --- Perez, 1990

\textbf{Levelized Cost} (LCOE): $LCOE = (I + \sum M_t/(1+r)^t)/(\sum E_t/(1+r)^t)$ --- IEA/NREL


\section{Derivation Chain}

\subsection*{Step 1: Solar irradiance}
\textbf{Inputs:} $\beta, \gamma, lat$

$G = G_{DN}\cos\theta + G_{DH}(1+\cos\beta)/2$

\textbf{Principle:} Perez model --- Tilted

\subsection*{Step 2: PV output}
\textbf{Inputs:} $G, P_{pv}, \eta$

$P_{pv,out} = P_{pv} \times G/1000 \times \eta_{sys}$

\textbf{Principle:} STC --- DC power

\subsection*{Step 3: AC output}
\textbf{Inputs:} $P_{pv,out}, P_{inv}$

$P_{ac} = \min(P_{pv,out}, P_{inv}) \times \eta_{inv}$

\textbf{Principle:} Clipping --- Grid power

\subsection*{Step 4: Net load}
\textbf{Inputs:} $P_{ac}, L$

$P_{net} = L - P_{ac}$

\textbf{Principle:} Balance --- Deficit/surplus

\subsection*{Step 5: Battery dispatch}
\textbf{Inputs:} $P_{net}, SOC$

$P_b = clip(P_{net}, SOC constraints)$

\textbf{Principle:} Limits --- Charge/discharge

\subsection*{Step 6: Unmet load}
\textbf{Inputs:} $P_{net}, P_b$

$P_{unmet} = max(0, P_{net} - P_b)$

\textbf{Principle:} Deficit --- Reliability

\subsection*{Step 7: LOLP}
\textbf{Inputs:} $P_{unmet}, L$

$LOLP = \sum P_{unmet} / \sum L$

\textbf{Principle:} Definition --- Constraint

\subsection*{Step 8: LCOE}
\textbf{Inputs:} $CAPEX, OPEX, E$

$LCOE = (CAPEX + \sum OPEX/(1+r)^t) / \sum E/(1+r)^t$

\textbf{Principle:} Definition --- Minimize
