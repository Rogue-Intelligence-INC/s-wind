\begin{abstract}
This paper presents exhaustive parametric optimization for renewable microgrid configuration using the S-wind simulation framework.
We systematically explore a search space of 8,820 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Minimize cost of energy with 100% renewable penetration}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\textbf{Capacity Factor} (CAPACITY_FACTOR): $CF = E_{actual} / (P_{rated} \times T)$ --- Definition

\textbf{Levelized Cost} (LCOE): $LCOE = (CRF \times CAPEX + OPEX) / E$ --- Energy Economics


\section{Derivation Chain}

\subsection*{Step 1: Solar generation}
\textbf{Inputs:} $f_s, P_{peak}, CF_s$

$E_s = f_s P_{peak} CF_s \times 8760$

\textbf{Principle:} Annual --- Solar energy

\subsection*{Step 2: Wind generation}
\textbf{Inputs:} $f_w, P_{peak}, CF_w$

$E_w = f_w P_{peak} CF_w \times 8760$

\textbf{Principle:} Annual --- Wind energy

\subsection*{Step 3: Total generation}
\textbf{Inputs:} $E_s, E_w$

$E_{gen} = E_s + E_w$

\textbf{Principle:} Sum --- Renewable

\subsection*{Step 4: Net load}
\textbf{Inputs:} $L, E_{gen}$

$L_{net}(t) = L(t) - E_{gen}(t)$

\textbf{Principle:} Residual --- Mismatch

\subsection*{Step 5: Storage dispatch}
\textbf{Inputs:} $L_{net}, E_b$

$P_b(t) = f(L_{net}, SOC)$

\textbf{Principle:} Algorithm --- Arbitrage

\subsection*{Step 6: Unmet demand}
\textbf{Inputs:} $L_{net}, P_b$

$U = \int \max(0, L_{net} - P_b) dt$

\textbf{Principle:} Deficit --- Reliability

\subsection*{Step 7: Curtailment}
\textbf{Inputs:} $E_{gen}, L, P_b$

$C = \int \max(0, E_{gen} - L - P_{charge}) dt$

\textbf{Principle:} Excess --- Waste

\subsection*{Step 8: Total cost}
\textbf{Inputs:} $CAPEX, OPEX$

$COE = (CAPEX_{ann} + OPEX) / E_{served}$

\textbf{Principle:} LCOE --- Minimize
