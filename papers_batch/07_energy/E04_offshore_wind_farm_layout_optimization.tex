\begin{abstract}
This paper presents exhaustive parametric optimization for offshore wind farm layout optimization using the S-wind simulation framework.
We systematically explore a search space of 140,800 configurations through physics-based modeling
combined with semi-empirical correlations. The objective function is: \textbf{Maximize annual energy production considering wake effects}.
Complete derivation chains from first principles are provided, enabling full reproducibility.
\end{abstract}

\section{Theoretical Foundation}

\subsection{Knowledge Base References}
\textbf{Jensen Wake Model} (JENSEN): $u/U = 1 - (1-\sqrt{1-C_T})/(1+2kx/D)^2$ --- Jensen, 1983

\textbf{Betz Limit} (BETZ): $C_{p,max} = 16/27 \approx 0.593$ --- Betz, 1919


\section{Derivation Chain}

\subsection*{Step 1: Wake deficit}
\textbf{Inputs:} $C_T, x, D$

$\frac{\Delta U}{U_0} = \frac{1-\sqrt{1-C_T}}{(1+2kx/D)^2}$

\textbf{Principle:} Jensen --- Velocity reduction

\subsection*{Step 2: Wake expansion}
\textbf{Inputs:} $k, x$

$D_w = D + 2kx$

\textbf{Principle:} Linear --- Wake diameter

\subsection*{Step 3: Partial wake}
\textbf{Inputs:} $overlap$

$A_{wake} = D_w^2 \arccos(...)$

\textbf{Principle:} Geometry --- Area fraction

\subsection*{Step 4: Effective velocity}
\textbf{Inputs:} $U_0, wakes$

$U_{eff} = U_0(1 - \sqrt{\sum(\Delta U/U_0)^2})$

\textbf{Principle:} Superposition --- Combined

\subsection*{Step 5: Power output}
\textbf{Inputs:} $U_{eff}, \rho, A, C_p$

$P = 0.5 \rho A C_p U_{eff}^3$

\textbf{Principle:} Betz --- Per turbine

\subsection*{Step 6: Farm power}
\textbf{Inputs:} $P_i$

$P_{farm} = \sum_i P_i$

\textbf{Principle:} Sum --- Total

\subsection*{Step 7: Capacity factor}
\textbf{Inputs:} $P_{farm}, P_{rated}, N$

$CF = P_{farm}/(N \times P_{rated})$

\textbf{Principle:} Definition --- Utilization

\subsection*{Step 8: AEP}
\textbf{Inputs:} $P_{farm}, hours$

$AEP = \sum_{bins} P(U) f(U) \times 8760$

\textbf{Principle:} Weibull --- Annual energy
