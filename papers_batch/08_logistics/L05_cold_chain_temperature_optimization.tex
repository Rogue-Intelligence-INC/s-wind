\section{Knowledge Base}
\textbf{Fourier Law} (FOURIER): $q = -k \nabla T$ --- Heat conduction

\textbf{Carnot COP} (CARNOT_COP): $COP = T_c/(T_h - T_c)$ --- Thermodynamics


\section{Derivation Chain}
\textbf{Step 1}: Heat infiltration --- Inputs: $U, A, \Delta T$, Formula: $Q_{inf} = U A (T_a - T_s)$, Principle: Conduction (W)


\textbf{Step 2}: Door losses --- Inputs: $t_o, Q_{door}$, Formula: $Q_d = t_o \times k_{door} (T_a - T_s)$, Principle: Air exchange (W)


\textbf{Step 3}: Product load --- Inputs: $m, c_p, \dot{T}$, Formula: $Q_p = m c_p \dot{T}$, Principle: Cooling (W)


\textbf{Step 4}: Total heat load --- Inputs: $Q$, Formula: $Q_{total} = Q_{inf} + Q_d + Q_p$, Principle: Sum (W)


\textbf{Step 5}: COP --- Inputs: $T_s, T_a$, Formula: $COP = (T_s + 273)/(T_a - T_s)$, Principle: Carnot (Efficiency)


\textbf{Step 6}: Power consumption --- Inputs: $Q, COP$, Formula: $P = Q_{total}/COP$, Principle: Definition (kW)


\textbf{Step 7}: Energy cost --- Inputs: $P, t, rate$, Formula: $C = P \times t \times rate$, Principle: Operation ($/day)


\textbf{Step 8}: Quality metric --- Inputs: $T, t$, Formula: $Q_f = e^{-k(T-T_{ref})t}$, Principle: Arrhenius (Shelf life)
