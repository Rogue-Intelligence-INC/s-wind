\section{Knowledge Base}
\textbf{Spatial Poisson} (SPATIAL_POISSON): $P(n,A) = (\lambda A)^n e^{-\lambda A}/n!$ --- Point process

\textbf{Logistic Choice} (LOGIT): $P = 1/(1+e^{-\beta x})$ --- McFadden, 1974


\section{Derivation Chain}
\textbf{Step 1}: Driver availability --- Inputs: $N, \rho$, Formula: $A = N \times (1-\rho)$, Principle: Active drivers (Available)


\textbf{Step 2}: Matching probability --- Inputs: $A, \lambda, r$, Formula: $P_m = 1 - e^{-\pi r^2 \rho_d}$, Principle: Spatial Poisson (Within radius)


\textbf{Step 3}: Wait time --- Inputs: $P_m, \lambda$, Formula: $W = 1/(\lambda P_m)$, Principle: Queueing (Expected)


\textbf{Step 4}: Surge trigger --- Inputs: $W, W_{max}$, Formula: $s = 1 + \max(0, (W/W_{max}-1))$, Principle: Threshold (Price increase)


\textbf{Step 5}: Driver acceptance --- Inputs: $p, s$, Formula: $P_a = \Phi((p \times s - c)/\sigma)$, Principle: Logit (Probability)


\textbf{Step 6}: Fulfilled orders --- Inputs: $\lambda, P_m, P_a$, Formula: $F = \lambda P_m P_a$, Principle: Product (Per hour)


\textbf{Step 7}: Cost per delivery --- Inputs: $p_b, s, F$, Formula: $C = p_b \times s$, Principle: Variable (Payment)


\textbf{Step 8}: SLA metric --- Inputs: $W, W_{max}$, Formula: $SLA = P(W \leq W_{max})$, Principle: Probability (Constraint)
