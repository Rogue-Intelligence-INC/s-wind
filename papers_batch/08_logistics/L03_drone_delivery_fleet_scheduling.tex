\section{Knowledge Base}
\textbf{Multirotor Power} (DRONE_POWER): $P = T^{3/2}/\sqrt{2\rho A}$ --- Momentum theory

\textbf{Battery Discharge} (BATTERY): $E = \int P(t) dt$ --- Integration


\section{Derivation Chain}
\textbf{Step 1}: Flight range --- Inputs: $E, P_{flight}$, Formula: $R = E v / P_{flight}$, Principle: Definition (km)


\textbf{Step 2}: Flight power --- Inputs: $m, v$, Formula: $P = k_1 + k_2 m + k_3 v^3$, Principle: Aerodynamics (W)


\textbf{Step 3}: Trip time --- Inputs: $d, v$, Formula: $t_{trip} = 2d/v + t_{load}$, Principle: Round trip (minutes)


\textbf{Step 4}: Charge time --- Inputs: $E, P_c$, Formula: $t_{charge} = E / P_c$, Principle: Definition (hours)


\textbf{Step 5}: Trips per charge --- Inputs: $R, d_{avg}$, Formula: $n_{trips} = R / (2 d_{avg})$, Principle: Integer (Before recharge)


\textbf{Step 6}: Daily capacity --- Inputs: $N_d, trips$, Formula: $C_{day} = N_d \times trips \times T_{op}/t_{cycle}$, Principle: Aggregation (Deliveries)


\textbf{Step 7}: Airspace slots --- Inputs: $region$, Formula: $S = f(traffic, regulations)$, Principle: Constraint (Available windows)


\textbf{Step 8}: Throughput --- Inputs: $C_{day}, S$, Formula: $Q = \min(C_{day}, S)$, Principle: Bottleneck (Actual)
