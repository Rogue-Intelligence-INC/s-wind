\section{Derivation Chain}
\begin{longtable}{|c|p{2cm}|p{2cm}|p{3cm}|p{2cm}|p{3cm}|}
\hline
Step & Desc & In & Out & Rule & Why\\\hline\endhead
1 & GEMM FLOPs & $M, N, K$ & $F = 2MNK$ & Definition & Matrix multiply\\\hline

3 & Reuse factor & $M, N, K, T_*$ & $R = MNK/(tiles×tile_size)$ & Reuse & Data reuse ratio\\\hline
4 & DRAM accesses & $F, R$ & $A_dram = F / R$ & Traffic & Off-chip accesses\\\hline
5 & Energy compute & $F, E_mac$ & $E_comp = F × E_mac$ & Compute & MAC energy\\\hline
6 & Energy memory & $A_dram, E_dram$ & $E_mem = A_dram × E_dram$ & Memory & DRAM energy\\\hline
7 & Total energy & $E_comp, E_mem$ & $E = E_comp + E_mem$ & Sum & Total energy\\\hline
8 & Latency & $F, P, f$ & $T = F / (P × f)$ & Parallelism & Execution time\\\hline
9 & EDP & $E, T$ & $EDP = E × T$ & Objective & Energy-delay product\\\hline
\end{longtable}
