\section{Problem}
\textbf{Objective}: Minimize latency × power / throughput

\section{Derivation Chain}
\begin{longtable}{|c|p{2cm}|p{2cm}|p{3cm}|p{2cm}|p{3cm}|}
\hline
Step & Desc & In & Out & Rule & Why\\\hline\endhead
1 & Model memory & $N, P$ & $M_model = N×P/8$ & Definition & Bits to bytes\\\hline
2 & KV cache & $B, L, d, n$ & $M_kv = 2BLdn×P/8$ & Architecture & K and V per layer\\\hline
3 & Compression & $M_kv, r$ & $M_kv' = M_kv×r$ & Compression & Quantization\\\hline
4 & Total memory & $M_m, M_kv, TP$ & $M = M_m/TP + M_kv'$ & Sharding & Model sharded\\\hline
5 & FLOPs/token & $N$ & $F = 2N$ & Standard & ~2 FLOPs/param\\\hline
6 & Compute throughput & $F, TFLOPS, B$ & $T_c = TFLOPS×10¹²×B/F$ & Roofline & Compute bound\\\hline
7 & Memory throughput & $M, BW, B$ & $T_m = BW×B/M$ & Bandwidth & Memory bound\\\hline
8 & Effective throughput & $T_c, T_m$ & $T = min(T_c, T_m)$ & Bottleneck & Slower dominates\\\hline
\end{longtable}
