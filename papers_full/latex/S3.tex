\section{Derivation Chain}
\begin{longtable}{|c|p{2cm}|p{2cm}|p{3cm}|p{2cm}|p{3cm}|}
\hline
Step & Desc & In & Out & Rule & Why\\\hline\endhead
1 & Fiber reach & $r_fiber, HH$ & $HH_fiber = r_fiber × HH_total$ & Definition & Fiber homes\\\hline
2 & Wireless reach & $D_tower, A$ & $HH_wireless = f(D_tower, A, terrain)$ & Propagation & Cell coverage\\\hline

5 & Tower cost & $D_tower, A$ & $C_tower = D_tower × A × c_tower$ & Deployment & Capital cost\\\hline
6 & Operating cost & $C_*, t$ & $OpEx = (C_fiber + C_tower) × r_opex × t$ & O\&M & Annual cost\\\hline
7 & Cost per HH & $Total, HH_served$ & $$/HH = Total / HH_served$ & Unit cost & Economics\\\hline
\end{longtable}
