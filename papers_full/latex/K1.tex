\section{Problem}
\textbf{Objective}: Minimize total vehicle delay

\section{Derivation Chain}
\begin{longtable}{|c|p{2cm}|p{2cm}|p{3cm}|p{2cm}|p{3cm}|}
\hline
Step & Desc & In & Out & Rule & Why\\\hline\endhead
1 & Saturation flow & $lanes, s_0$ & $s = lanes × s_0$ & Definition & veh/hr green\\\hline
2 & Capacity & $s, g, C$ & $c = s × g/C$ & Webster & veh/hr\\\hline
3 & Degree of saturation & $q, c$ & $x = q/c$ & Definition & v/c ratio\\\hline
4 & Uniform delay & $C, g, x$ & $d_1 = C(1-g/C)²/(2(1-x×g/C))$ & Webster & Signal delay\\\hline
5 & Random delay & $x, c$ & $d_2 = x²/(2c(1-x))$ & Queuing & Overflow delay\\\hline
6 & Total delay & $d_1, d_2$ & $d = d_1 + d_2$ & Sum & Per vehicle\\\hline
7 & Network delay & $d_i, q_i$ & $D = Σ d_i × q_i$ & Aggregation & Total network\\\hline
\end{longtable}
