\section{Problem}

\section{Derivation Chain}
\begin{longtable}{|c|p{2cm}|p{2cm}|p{3cm}|p{2cm}|p{3cm}|}
\hline
Step & Desc & In & Out & Rule & Why\\\hline\endhead
1 & Volume & $D, L$ & $V = π(D/2)²×L$ & Geometry & Cylinder volume\\\hline
2 & Shell mass & $D, L, t, ρ$ & $M_shell = π×D×L×t×ρ$ & Geometry & Cylinder surface\\\hline
3 & Insulation mass & $A, R$ & $M_ins = A × k/R × ρ_ins$ & Thermal & Conductivity relation\\\hline
4 & Shielding mass & $A, ρ_s$ & $M_shield = A × ρ_s$ & Definition & Areal density\\\hline
5 & Total mass & $M_*$ & $M = M_shell + M_ins + M_shield$ & Sum & All components\\\hline
6 & Heat loss & $A, ΔT, R$ & $Q = A × ΔT / R$ & Fourier & Steady-state\\\hline
7 & Radiation dose & $ρ_s, GCR$ & $dose = GCR × exp(-ρ_s/λ)$ & Exponential & Attenuation\\\hline
8 & Objective & $V, M$ & $V/M$ & Efficiency & Volume per mass\\\hline
\end{longtable}
